                    self.page.wait_for_timeout(300)

                    break

                break

            cancel = discard.locator("button").filter(has_text=re.compile(r"取\s*消"))
            if cancel.count() > 0:
                cancel.first.click(timeout=3000)
                self.page.wait_for_timeout(400)
